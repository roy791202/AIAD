class GitWorkbenchSignals(QObject): # 狀態與過濾信號 status_filter_changed = Signal(str) # 'gray', 'yellow', 'red', 'green', 'all' task_selected = Signal(str) # taskId (e.g. 'TASK-01') file_selection_changed = Signal(list, list) # staged_files, unstaged_files # Git 操作請求 commit_requested = Signal(str, list) # commit_message, files sync_requested = Signal(str) # branch_name branch_switched = Signal(str) # target_branch # 背景背景任務回傳 (非同步) git_operation_started = Signal(str) # operation_name git_operation_finished = Signal(bool, str) # success, message diff_loaded = Signal(str, str) # file_path, diff_content
