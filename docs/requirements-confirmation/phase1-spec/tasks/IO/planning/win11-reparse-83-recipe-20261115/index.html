ver
systeminfo | findstr /B /C:"OS Name" /C:"OS Version"
wmic os get Caption,Version,BuildNumber,OSArchitecture

set "ROOT=%TEMP%\io-fx-reparse-%RANDOM%"
mkdir "%ROOT%\real-target"
echo payload> "%ROOT%\real-target\file.bin"
mklink /J "%ROOT%\junction-alias" "%ROOT%\real-target"
fsutil reparsepoint query "%ROOT%\junction-alias"
rmdir "%ROOT%\junction-alias"
rmdir /s /q "%ROOT%"

fsutil 8dot3name query %TEMP%
set "ROOT=%TEMP%\io-fx-83-%RANDOM%"

echo payload> "%ROOT%\LongFileNameForShortAlias.bin"
dir /x "%ROOT%"
fsutil file queryShortName "%ROOT%\LongFileNameForShortAlias.bin"
